    SELECT
      assay_flags.assay_name,
      COUNT(*) AS flagged_readings
    FROM lead_opt_demo.silver.assay_flags
    WHERE assay_flags.is_flagged = true
      AND assay_flags.assay_name IS NOT NULL
    GROUP BY assay_flags.assay_name

    SELECT
      flagged_counts.assay_name,
      flagged_counts.flagged_readings,

  SELECT
    ranked.assay_name,
    ranked.flagged_readings
